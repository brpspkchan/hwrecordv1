    // =========================================================
    // 模組 1: 快速登記 (Data Grid)
    // =========================================================
    function RegisterModule({ classes, students, records, setRecords, sheetUrl, showToast }) {
      const [date, setDate] = useState(getTodayStr());
      const [selectedClass, setSelectedClass] = useState(classes[0]?.id || "");
      const [subject, setSubject] = useState(SUBJECTS[0]);
      const [hwName, setHwName] = useState("");
      const [isSyncing, setIsSyncing] = useState(false);

      const [focusedRowIndex, setFocusedRowIndex] = useState(0);
      // 本地表單狀態: Map<studentId, boolean(isMissing)>
      const [missingMap, setMissingMap] = useState({});

      // 取得目前班別的所有啟用學生 (按座號排序)
      const currentClassStudents = useMemo(() => {
        return students
          .filter(s => s.classId === selectedClass && s.status === 'active')
          .sort((a, b) => a.seatNum - b.seatNum);
      }, [students, selectedClass]);

      // 當 [日期, 班別, 科目] 改變時，自動從既有紀錄中讀取已儲存的狀態
      useEffect(() => {
        const existingRecords = records.filter(
          r => r.date === date && r.classId === selectedClass && r.subject === subject
        );

        const newMap = {};
        // 預設全部為已交 (false)
        currentClassStudents.forEach(s => {
          newMap[s.id] = false;
        });

        if (existingRecords.length > 0) {
          // 如果當天該科已有名單紀錄，還原作業名稱與欠交狀態
          setHwName(existingRecords[0].hwName || "");
          existingRecords.forEach(r => {
            newMap[r.studentId] = r.isMissing;
          });
        } else {
          setHwName("");
        }

        setMissingMap(newMap);
        setFocusedRowIndex(0);
      }, [date, selectedClass, subject, currentClassStudents, records]);

      // 切換單個學生的欠交狀態
      const toggleMissing = (studentId) => {
        setMissingMap(prev => ({
          ...prev,
          [studentId]: !prev[studentId]
        }));
      };

      // 重置全班狀態為已交
      const handleSelectAllSubmitted = () => {
        const newMap = {};
        currentClassStudents.forEach(s => {
          newMap[s.id] = false;
        });
        setMissingMap(newMap);
      };

      // 鍵盤快捷鍵極速操作 (Space: 切換勾選, Down: 下一位, Up: 上一位)
      useEffect(() => {
        const handleKeyDown = (e) => {
          // 若使用者正在輸入作業名稱文字框，不觸發表格鍵盤導航
          if (document.activeElement.tagName === 'INPUT' && document.activeElement.type === 'text') {
            return;
          }

          if (e.code === 'Space') {
            e.preventDefault();
            if (currentClassStudents[focusedRowIndex]) {
              toggleMissing(currentClassStudents[focusedRowIndex].id);
            }
          } else if (e.code === 'ArrowDown') {
            e.preventDefault();
            setFocusedRowIndex(prev => Math.min(prev + 1, currentClassStudents.length - 1));
          } else if (e.code === 'ArrowUp') {
            e.preventDefault();
            setFocusedRowIndex(prev => Math.max(prev - 1, 0));
          }
        };

        window.addEventListener('keydown', handleKeyDown);
        return () => window.removeEventListener('keydown', handleKeyDown);
      }, [focusedRowIndex, currentClassStudents]);

      // 儲存紀錄與即時同步
      const handleSave = async () => {
        // 移除舊的相同 [日期 + 班別 + 科目] 紀錄
        const filteredRecords = records.filter(
          r => !(r.date === date && r.classId === selectedClass && r.subject === subject)
        );

        // 建立新紀錄（只寫入該班現有學生）
        const newEntries = currentClassStudents.map(s => ({
          id: `${date}_${selectedClass}_${subject}_${s.id}`,
          date,
          classId: selectedClass,
          subject,
          hwName: hwName.trim(), // 預留空白，允許為空
          studentId: s.id,
          isMissing: !!missingMap[s.id]
        }));

        const updatedRecords = [...filteredRecords, ...newEntries];
        setRecords(updatedRecords);

        // 若已綁定 Google Sheet URL，即時同步欠交資料 (A行：欠交學生姓名, B行：欠交日期)
        if (sheetUrl) {
          setIsSyncing(true);
          const missingEntries = newEntries
            .filter(r => r.isMissing)
            .map(r => {
              const student = students.find(s => s.id === r.studentId);
              return {
                studentName: student ? `${student.nameZh} (${selectedClass})` : "未知學生",
                date: r.date,
                className: selectedClass,
                subject: r.subject,
                hwName: r.hwName || ''
              };
            });

          try {
            await fetch(sheetUrl, {
              method: 'POST',
              mode: 'no-cors',
              headers: { 'Content-Type': 'application/json' },
              body: JSON.stringify({ action: 'append', records: missingEntries })
            });
            showToast(`已儲存並成功即時同步 ${missingEntries.length} 筆欠交紀錄至 Google Sheet！`);
          } catch (err) {
            console.error(err);
            showToast(`本地已儲存，但 Google Sheet 同步失敗。`, 'error');
          } finally {
            setIsSyncing(false);
          }
        } else {
          showToast(`已成功儲存！(提示：可至 Admin 頁面綁定 Google Sheet 啟用即時同步)`);
        }
      };

      const missingCount = Object.values(missingMap).filter(Boolean).length;

      return (
        <div class="space-y-6">
          {/* 控制表頭卡片 */}
          <div class="bg-white p-5 rounded-2xl shadow-sm border border-slate-200">
            <h2 class="text-lg font-bold text-slate-800 mb-4 flex items-center space-x-2">
              <i data-lucide="sliders" class="w-5 h-5 text-indigo-600"></i>
              <span>登記參數設定</span>
            </h2>
            <div class="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
              <div>
                <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">日期</label>
                <input
                  type="date"
                  value={date}
                  onChange={(e) => setDate(e.target.value)}
                  class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:bg-white outline-none transition"
                />
              </div>

              <div>
                <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">班別</label>
                <select
                  value={selectedClass}
                  onChange={(e) => setSelectedClass(e.target.value)}
                  class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:bg-white outline-none transition"
                >
                  {classes.map(c => (
                    <option key={c.id} value={c.id}>{c.name} 班 (班主任: {c.teacher || '未設定'})</option>
                  ))}
                </select>
              </div>

              <div>
                <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">科目</label>
                <select
                  value={subject}
                  onChange={(e) => setSubject(e.target.value)}
                  class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:bg-white outline-none transition"
                >
                  {SUBJECTS.map(s => (
                    <option key={s} value={s}>{s}</option>
                  ))}
                </select>
              </div>

              <div>
                <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">
                  作業名稱 <span class="text-slate-400 font-normal">(可空白)</span>
                </label>
                <input
                  type="text"
                  placeholder="例如：單元三工作紙"
                  value={hwName}
                  onChange={(e) => setHwName(e.target.value)}
                  class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm focus:ring-2 focus:ring-indigo-500 focus:bg-white outline-none transition"
                />
              </div>
            </div>
          </div>

          {/* 操作說明與統計提示 */}
          <div class="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-3 bg-indigo-50 p-4 rounded-xl border border-indigo-100">
            <div class="flex items-center space-x-2 text-xs text-indigo-800">
              <i data-lucide="keyboard" class="w-4 h-4 text-indigo-600"></i>
              <span>快捷鍵小貼士：按 <kbd class="px-1.5 py-0.5 bg-white border border-indigo-200 rounded font-mono shadow-sm">Space</kbd> 切換欠交狀態，按 <kbd class="px-1.5 py-0.5 bg-white border border-indigo-200 rounded font-mono shadow-sm">↓</kbd> <kbd class="px-1.5 py-0.5 bg-white border border-indigo-200 rounded font-mono shadow-sm">↑</kbd> 鍵移動學生。</span>
            </div>
            <div class="flex items-center space-x-4">
              <span class="text-xs font-semibold text-slate-600">
                欠交人數：<strong class="text-rose-600 text-sm font-bold">{missingCount}</strong> / {currentClassStudents.length} 人
              </span>
              <button
                onClick={handleSelectAllSubmitted}
                class="text-xs font-medium text-indigo-700 hover:text-indigo-900 hover:underline"
              >
                重置全班為已交
              </button>
            </div>
          </div>

          {/* 表格主體 (Data Grid) */}
          <div class="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
            <div class="overflow-x-auto max-h-[500px] custom-scrollbar">
              <table class="w-full text-left border-collapse">
                <thead class="bg-slate-100 sticky top-0 z-10 border-b border-slate-200 text-xs font-bold text-slate-600 uppercase tracking-wider">
                  <tr>
                    <th class="py-3 px-4 w-16 text-center">座號</th>
                    <th class="py-3 px-4">中文姓名</th>
                    <th class="py-3 px-4">英文姓名</th>
                    <th class="py-3 px-4 w-40 text-center">狀態 (點擊切換)</th>
                  </tr>
                </thead>
                <tbody class="divide-y divide-slate-100 text-sm">
                  {currentClassStudents.length === 0 ? (
                    <tr>
                      <td colSpan="4" class="py-8 text-center text-slate-400">該班別暫無學生資料，請先至 Admin 管理頁面新增學生。</td>
                    </tr>
                  ) : (
                    currentClassStudents.map((s, idx) => {
                      const isMissing = !!missingMap[s.id];
                      const isFocused = focusedRowIndex === idx;

                      return (
                        <tr
                          key={s.id}
                          onClick={() => setFocusedRowIndex(idx)}
                          class={`transition cursor-pointer ${
                            isFocused ? 'bg-indigo-50/70 ring-1 ring-inset ring-indigo-300' : 'hover:bg-slate-50'
                          }`}
                        >
                          <td class="py-3 px-4 text-center font-medium text-slate-500">{s.seatNum}</td>
                          <td class="py-3 px-4 font-semibold text-slate-800">{s.nameZh}</td>
                          <td class="py-3 px-4 text-slate-500 font-mono text-xs">{s.nameEn}</td>
                          <td class="py-3 px-4 text-center">
                            <button
                              onClick={(e) => {
                                e.stopPropagation();
                                toggleMissing(s.id);
                              }}
                              class={`px-4 py-1.5 rounded-full text-xs font-bold transition-all shadow-sm ${
                                isMissing
                                  ? 'bg-rose-500 hover:bg-rose-600 text-white shadow-rose-200'
                                  : 'bg-emerald-100 hover:bg-emerald-200 text-emerald-800'
                              }`}
                            >
                              {isMissing ? '欠交' : '已交'}
                            </button>
                          </td>
                        </tr>
                      );
                    })
                  )}
                </tbody>
              </table>
            </div>

            {/* 提交按鈕區域 */}
            <div class="p-4 bg-slate-50 border-t border-slate-200 flex items-center justify-between">
              <div class="text-xs text-slate-500 flex items-center space-x-1.5">
                <span class={`w-2 h-2 rounded-full ${sheetUrl ? 'bg-emerald-500' : 'bg-amber-500'}`}></span>
                <span>{sheetUrl ? '已連線至 Google Sheet (即時同步中)' : '尚未綁定 Google Sheet 網址'}</span>
              </div>
              <button
                onClick={handleSave}
                disabled={isSyncing}
                class="px-6 py-3 bg-indigo-600 hover:bg-indigo-700 disabled:bg-slate-400 text-white font-bold rounded-xl shadow-lg hover:shadow-indigo-200 transition flex items-center space-x-2"
              >
                <i data-lucide={isSyncing ? "refresh-cw" : "save"} class={`w-5 h-5 ${isSyncing ? 'animate-spin' : ''}`}></i>
                <span>{isSyncing ? '同步中...' : '儲存並同步至 Google Sheet'}</span>
              </button>
            </div>
          </div>
        </div>
      );
    }

    function SummaryModule({ classes, students, records }) {
      const [filterClass, setFilterClass] = useState("ALL");
      const [filterSubject, setFilterSubject] = useState("ALL");
      const [searchQuery, setSearchQuery] = useState("");

      // 核心邏輯：計算累計欠交
      // 規則 1: 跨科獨立算次數
      // 規則 2: 「當天同一學生無論欠交多少功課，歷史累計欠交天數只記錄為一次」
      const summaryList = useMemo(() => {
        return students.map(student => {
          // 該生的所有欠交紀錄 (isMissing = true)
          const studentMissingRecords = records.filter(
            r => r.studentId === student.id && r.isMissing
          );

          // 科目過濾條件下的紀錄
          const filteredRecords = studentMissingRecords.filter(r => {
            if (filterSubject !== "ALL" && r.subject !== filterSubject) return false;
            return true;
          });

          // 1. 各科欠交統計 Map { 中文: 2, 英文: 1 }
          const subjectBreakdown = {};
          filteredRecords.forEach(r => {
            subjectBreakdown[r.subject] = (subjectBreakdown[r.subject] || 0) + 1;
          });

          // 2. 總欠交項目數
          const totalItemsCount = filteredRecords.length;

          // 3. 歷史累計不重複欠交天數
          const uniqueDays = new Set(filteredRecords.map(r => r.date)).size;

          const studentClass = classes.find(c => c.id === student.classId);

          return {
            ...student,

            classTeacher: studentClass ? studentClass.teacher : '未設定',
            subjectBreakdown,

        }).filter(item => {

              item.nameZh.toLowerCase().includes(query) ||
              item.nameEn.toLowerCase().includes(query) ||
              item.className.toLowerCase().includes(query)
            );
          }
          return true;
        });
      }, [students, records, classes, filterClass, filterSubject, searchQuery]);

        }));

        summaryList.forEach(s => {

            detailData.push({
              "日期": r.date,

              "座號": s.seatNum,
              "學生姓名": s.nameZh,
              "科目": r.subject,
              "作業名稱": r.hwName || '(無名稱)'
            });
          });
        });

        const wb = XLSX.utils.book_new();
        const wsOverview = XLSX.utils.json_to_sheet(overviewData);
        const wsDetail = XLSX.utils.json_to_sheet(detailData);

        XLSX.utils.book_append_sheet(wb, wsOverview, "欠交累計總覽");
        XLSX.utils.book_append_sheet(wb, wsDetail, "詳細紀錄明細");

        XLSX.writeFile(wb, `全校學生功課欠交統計報表_${getTodayStr()}.xlsx`);
      };

                  onChange={(e) => setFilterSubject(e.target.value)}

              <span>一鍵匯出 Excel 報表</span>

          {/* 統計列表 */}
          <div class="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">

    function AdminModule({ classes, setClasses, students, setStudents, records, setRecords, sheetUrl, setSheetUrl, showToast }) {
      const [adminTab, setAdminTab] = useState("gsheet"); // "gsheet" | "students" | "classes" | "promotion"
      
      // 新增班別表單
      const [newClassName, setNewClassName] = useState("");
      const [newClassTeacher, setNewClassTeacher] = useState("");

      // 新增單筆學生表單
      const [studentClassId, setStudentClassId] = useState(classes[0]?.id || "");
      const [studentNameZh, setStudentNameZh] = useState("");
      const [studentNameEn, setStudentNameEn] = useState("");

      // CSV 匯入預覽 Modal State
      const [csvPreviewData, setCsvPreviewData] = useState(null);

      // 手動全量覆蓋同步至 Google Sheet
      const [isFullSyncing, setIsFullSyncing] = useState(false);

      // CSV 下載範本
      const handleDownloadTemplate = () => {
        const csvContent = "data:text/csv;charset=utf-8,\uFEFF" + "班別,中文姓名,英文姓名\n1A,陳大文,Chan Tai Man\n1A,李小玲,Lee Siu Ling\n1B,張三,Zhang San";
        const encodedUri = encodeURI(csvContent);
        const link = document.createElement("a");
        link.setAttribute("href", encodedUri);
        link.setAttribute("download", "student_import_template.csv");
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
      };

      // CSV 上傳解析
      const handleFileUpload = (e) => {
        const file = e.target.files[0];
        if (!file) return;

        const reader = new FileReader();
        reader.onload = (event) => {
          const text = event.target.result;
          const lines = text.split(/\r\n|\n/).filter(line => line.trim() !== '');
          if (lines.length <= 1) {
            showToast("CSV 檔案內容為空或無有效學生資料！", "error");
            return;
          }

          const parsedStudents = [];
          const detectedClasses = new Set();

          for (let i = 1; i < lines.length; i++) {
            const cols = lines[i].split(',').map(c => c.trim().replace(/^["']|["']$/g, ''));
            if (cols.length >= 2 && cols[0] && cols[1]) {
              const classId = cols[0];
              const nameZh = cols[1];
              const nameEn = cols[2] || '';
              detectedClasses.add(classId);
              parsedStudents.push({ classId, nameZh, nameEn });
            }
          }

          if (parsedStudents.length === 0) {
            showToast("未從 CSV 解析到有效學生紀錄！", "error");
            return;
          }

          setCsvPreviewData({
            students: parsedStudents,
            detectedClasses: Array.from(detectedClasses)
          });
        };
        reader.readAsText(file);
      };

      // 確認 CSV 匯入
      const confirmCsvImport = () => {
        if (!csvPreviewData) return;

        // 1. 自動建立不存在的班別
        const existingClassIds = new Set(classes.map(c => c.id));
        const newClasses = [...classes];
        csvPreviewData.detectedClasses.forEach(cId => {
          if (!existingClassIds.has(cId)) {
            newClasses.push({ id: cId, name: cId, teacher: '' });
          }
        });
        setClasses(newClasses);

        // 2. 彙整學生並順序編排座號
        const newStudentList = [...students];

        csvPreviewData.students.forEach(s => {
          const classStudents = newStudentList.filter(st => st.classId === s.classId);
          const nextSeatNum = classStudents.length + 1;
          newStudentList.push({
            id: `s_${Date.now()}_${Math.random().toString(36).substr(2, 5)}`,
            classId: s.classId,
            seatNum: nextSeatNum,
            nameZh: s.nameZh,
            nameEn: s.nameEn,
            status: 'active'
          });
        });

        setStudents(newStudentList);
        showToast(`成功匯入 ${csvPreviewData.students.length} 名學生！`);
        setCsvPreviewData(null);
      };

      // 新增單筆學生
      const handleAddStudent = (e) => {
        e.preventDefault();
        if (!studentNameZh.trim() || !studentClassId) return;

        const classStudents = students.filter(s => s.classId === studentClassId);
        const nextSeatNum = classStudents.length + 1;

        const newStudent = {
          id: `s_${Date.now()}`,
          classId: studentClassId,
          seatNum: nextSeatNum,
          nameZh: studentNameZh.trim(),
          nameEn: studentNameEn.trim(),
          status: 'active'
        };

        setStudents([...students, newStudent]);
        setStudentNameZh("");
        setStudentNameEn("");
        showToast(`成功新增學生：${newStudent.nameZh}`);
      };

      // 刪除學生
      const handleDeleteStudent = (id) => {
        const target = students.find(s => s.id === id);
        if (!target) return;
        
        // 刪除後重新編排該班座號
        const remainingStudents = students.filter(s => s.id !== id);
        const updatedStudents = remainingStudents.map(s => {
          if (s.classId === target.classId && s.seatNum > target.seatNum) {
            return { ...s, seatNum: s.seatNum - 1 };
          }
          return s;
        });

        setStudents(updatedStudents);
        showToast(`已刪除學生：${target.nameZh}`);
      };

      // 新增班別
      const handleAddClass = (e) => {
        e.preventDefault();
        if (!newClassName.trim()) return;

        const classId = newClassName.trim().toUpperCase();
        if (classes.some(c => c.id === classId)) {

        };

        setNewClassName("");
        setNewClassTeacher("");

          id: promoteName(c.id),

          ...s,
          classId: promoteName(s.classId)
        }));

        setClasses(updatedClasses);
        setStudents(updatedStudents);
        setRecords([]);
        showToast("全校學年升級完成！舊欠交紀錄已重置。");
      };

      const appsScriptCode = `function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var payload = JSON.parse(e.postData.contents);
  var records = payload.records || [];

  if (payload.action === 'overwrite') {

    sheet.appendRow(['欠交學生姓名', '欠交日期', '班別', '科目', '功課名稱']);

      sheet.appendRow([r.studentName, r.date, r.className, r.subject, r.hwName]);

      sheet.appendRow(['欠交學生姓名', '欠交日期', '班別', '科目', '功課名稱']);
    }
    records.forEach(function(r) {
      sheet.appendRow([r.studentName, r.date, r.className, r.subject, r.hwName]);
    });
  }

  return ContentService.createTextOutput(JSON.stringify({ result: "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}`;

      const handleCopyCode = () => {

            </button>
            <button
              onClick={() => setAdminTab("classes")}
              class={`px-4 py-2 rounded-xl text-sm font-bold transition ${
                adminTab === 'classes' ? 'bg-indigo-600 text-white' : 'bg-white text-slate-600 hover:bg-slate-100'
              }`}
            >
              班別與班主任設定
            </button>
            <button
              onClick={() => setAdminTab("promotion")}
              class={`px-4 py-2 rounded-xl text-sm font-bold transition ${
                adminTab === 'promotion' ? 'bg-rose-600 text-white' : 'bg-white text-slate-600 hover:bg-slate-100'
              }`}

              學年升級與歸檔
            </button>
          </div>

          {/* 0. Google Sheet 即時同步設定 */}

                  </ol>
                </div>
              </div>
            </div>
          )}

          {/* 1. 學生資料管理 */}
          {adminTab === 'students' && (
            <div class="space-y-6">
              {/* CSV 批量匯入區 */}
              <div class="bg-white p-6 rounded-2xl shadow-sm border border-slate-200">
                <h3 class="text-base font-bold text-slate-800 mb-2 flex items-center space-x-2">
                  <i data-lucide="file-up" class="w-5 h-5 text-indigo-600"></i>
                  <span>CSV 批量匯入學生 (支援跨班)</span>
                </h3>
                <p class="text-xs text-slate-500 mb-4">
                  請上傳 CSV 檔案，格式必須包含：<strong class="text-slate-700">班別, 中文姓名, 英文姓名</strong>。系統將自動建立未存在的班別，並自動生成 1, 2, 3... 順序座號。
                </p>

                <div class="flex flex-col sm:flex-row items-stretch sm:items-center gap-3">
                  <input
                    type="file"
                    accept=".csv"
                    onChange={handleFileUpload}
                    class="block w-full text-sm text-slate-500 file:mr-4 file:py-2 py-1 file:px-4 file:rounded-xl file:border-0 file:text-xs file:font-semibold file:bg-indigo-50 file:text-indigo-700 hover:file:bg-indigo-100 cursor-pointer"
                  />
                  <button
                    onClick={handleDownloadTemplate}
                    class="px-4 py-2 bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-bold rounded-xl transition flex items-center justify-center space-x-1 shrink-0"
                  >
                    <i data-lucide="download" class="w-3.5 h-3.5"></i>
                    <span>下載 CSV 範本</span>
                  </button>
                </div>
              </div>

              {/* 手動新增單筆學生 */}
              <div class="bg-white p-6 rounded-2xl shadow-sm border border-slate-200">
                <h3 class="text-base font-bold text-slate-800 mb-4 flex items-center space-x-2">
                  <i data-lucide="user-plus" class="w-5 h-5 text-indigo-600"></i>
                  <span>手動新增單筆學生</span>
                </h3>
                <form onSubmit={handleAddStudent} class="grid grid-cols-1 sm:grid-cols-4 gap-4">
                  <div>
                    <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">班別</label>
                    <select
                      value={studentClassId}
                      onChange={(e) => setStudentClassId(e.target.value)}
                      class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm outline-none"
                    >
                      {classes.map(c => (
                        <option key={c.id} value={c.id}>{c.name} 班</option>
                      ))}
                    </select>
                  </div>
                  <div>
                    <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">中文姓名</label>
                    <input
                      type="text"
                      required
                      placeholder="例如：黃小明"
                      value={studentNameZh}
                      onChange={(e) => setStudentNameZh(e.target.value)}
                      class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm outline-none"
                    />
                  </div>
                  <div>
                    <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">英文姓名</label>
                    <input
                      type="text"
                      placeholder="Wong Siu Ming"
                      value={studentNameEn}
                      onChange={(e) => setStudentNameEn(e.target.value)}
                      class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm outline-none"
                    />
                  </div>
                  <div class="flex items-end">
                    <button
                      type="submit"
                      class="w-full py-2 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-xl shadow transition text-sm"
                    >
                      新增學生
                    </button>
                  </div>
                </form>
              </div>

              {/* 全校學生名單列表 */}
              <div class="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
                <div class="p-4 bg-slate-50 border-b border-slate-200 font-bold text-slate-700 text-sm">
                  全校學生名單 (座號自動順序重編)
                </div>
                <div class="overflow-x-auto max-h-[400px] custom-scrollbar">
                  <table class="w-full text-left border-collapse">
                    <thead class="bg-slate-100 text-xs font-bold text-slate-500 uppercase">
                      <tr>
                        <th class="py-2.5 px-4">班別</th>
                        <th class="py-2.5 px-4">座號</th>
                        <th class="py-2.5 px-4">中文姓名</th>
                        <th class="py-2.5 px-4">英文姓名</th>
                        <th class="py-2.5 px-4 text-center">操作</th>
                      </tr>
                    </thead>
                    <tbody class="divide-y divide-slate-100 text-sm">
                      {students.map(s => (
                        <tr key={s.id} class="hover:bg-slate-50">
                          <td class="py-2.5 px-4 font-semibold text-indigo-700">{s.classId}</td>
                          <td class="py-2.5 px-4 font-mono">{s.seatNum}</td>
                          <td class="py-2.5 px-4 font-bold text-slate-800">{s.nameZh}</td>
                          <td class="py-2.5 px-4 text-slate-500 font-mono text-xs">{s.nameEn}</td>
                          <td class="py-2.5 px-4 text-center">
                            <button
                              onClick={() => handleDeleteStudent(s.id)}
                              class="text-rose-600 hover:text-rose-800 text-xs font-bold hover:underline"
                            >
                              刪除
                            </button>
                          </td>
                        </tr>
                      ))}
                    </tbody>
                  </table>
                </div>
              </div>
            </div>
          )}

          {/* 2. 班別與班主任管理 */}
          {adminTab === 'classes' && (
            <div class="space-y-6">
              <div class="bg-white p-6 rounded-2xl shadow-sm border border-slate-200">
                <h3 class="text-base font-bold text-slate-800 mb-4">新增班別</h3>
                <form onSubmit={handleAddClass} class="grid grid-cols-1 sm:grid-cols-3 gap-4">
                  <div>
                    <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">班別名稱</label>
                    <input
                      type="text"
                      required
                      placeholder="例如：3A"
                      value={newClassName}
                      onChange={(e) => setNewClassName(e.target.value)}
                      class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm outline-none"
                    />
                  </div>
                  <div>
                    <label class="block text-xs font-semibold text-slate-500 uppercase tracking-wider mb-1">班主任</label>
                    <input
                      type="text"
                      placeholder="例如：歐陽老師"
                      value={newClassTeacher}
                      onChange={(e) => setNewClassTeacher(e.target.value)}
                      class="w-full bg-slate-50 border border-slate-300 rounded-xl px-3 py-2 text-sm outline-none"
                    />
                  </div>
                  <div class="flex items-end">
                    <button
                      type="submit"
                      class="w-full py-2 bg-indigo-600 hover:bg-indigo-700 text-white font-bold rounded-xl shadow transition text-sm"
                    >
                      新增班別
                    </button>
                  </div>
                </form>
              </div>

              <div class="bg-white rounded-2xl shadow-sm border border-slate-200 overflow-hidden">
                <div class="p-4 bg-slate-50 border-b border-slate-200 font-bold text-slate-700 text-sm">
                  現有班別列表
                </div>
                <div class="divide-y divide-slate-100">
                  {classes.map(c => (
                    <div key={c.id} class="p-4 flex justify-between items-center hover:bg-slate-50">
                      <div>
                        <span class="font-bold text-slate-800 text-base">{c.name} 班</span>
                        <span class="ml-4 text-xs text-slate-500">班主任：<strong class="text-slate-700">{c.teacher || '未設定'}</strong></span>
                      </div>
                      <span class="text-xs text-slate-400">
                        學生人數：{students.filter(s => s.classId === c.id).length} 人
                      </span>
                    </div>
                  ))}
                </div>
              </div>
            </div>
          )}

          {/* 3. 學年升級 */}
          {adminTab === 'promotion' && (
            <div class="bg-white p-6 rounded-2xl shadow-sm border border-slate-200 space-y-4">
              <h3 class="text-lg font-bold text-slate-800 flex items-center space-x-2">
                <i data-lucide="arrow-up-circle" class="w-5 h-5 text-rose-600"></i>
                <span>全校學年升級 (Promotion)</span>
              </h3>
              <p class="text-sm text-slate-600 leading-relaxed">
                點擊下方按鈕將執行全校升級程序：
                <br />1. 所有班別名稱與學生所屬班別的年級自動 +1（例如 1A 升為 2A）。
                <br />2. 清除舊學期的欠交功課紀錄並歸檔結算。
              </p>
              <div class="pt-2">
                <button
                  onClick={handlePromotion}
                  class="px-6 py-3 bg-rose-600 hover:bg-rose-700 text-white font-bold rounded-xl shadow-lg transition flex items-center space-x-2"
                >
                  <i data-lucide="zap" class="w-5 h-5"></i>
                  <span>一鍵執行全校學年升級</span>
                </button>
              </div>
            </div>
          )}

          {/* CSV 預覽 Modal */}
          {csvPreviewData && (
            <div class="fixed inset-0 bg-slate-900/50 backdrop-blur-sm z-50 flex items-center justify-center p-4">
              <div class="bg-white rounded-2xl max-w-2xl w-full p-6 space-y-4 shadow-2xl">
                <h3 class="text-lg font-bold text-slate-800">確認匯入 CSV 資料</h3>
                <p class="text-xs text-slate-500">
                  即將匯入 <strong class="text-indigo-600">{csvPreviewData.students.length}</strong> 名學生，包含以下班別：
                  {csvPreviewData.detectedClasses.join(', ')}。
                </p>

                <div class="max-h-60 overflow-y-auto border border-slate-200 rounded-xl p-2 text-xs font-mono bg-slate-50">
                  {csvPreviewData.students.map((s, idx) => (
                    <div key={idx} class="py-1 border-b border-slate-100 last:border-0">
                      {s.classId}班 - {s.nameZh} ({s.nameEn})
                    </div>
                  ))}
                </div>

                <div class="flex justify-end space-x-3 pt-2">
                  <button
                    onClick={() => setCsvPreviewData(null)}
                    class="px-4 py-2 bg-slate-100 hover:bg-slate-200 text-slate-700 text-xs font-bold rounded-xl"
                  >
                    取消
                  </button>
                  <button
                    onClick={confirmCsvImport}
                    class="px-5 py-2 bg-indigo-600 hover:bg-indigo-700 text-white text-xs font-bold rounded-xl shadow"
                  >
                    確認批量匯入
                  </button>
                </div>
              </div>
            </div>
          )}
        </div>
      );
    }

    // 渲染應用
    ReactDOM.render(<App />, document.getElementById('root'));
  </script>
