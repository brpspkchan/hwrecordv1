  <script>
    // Initialize icons
    lucide.createIcons();

    // Default Today's Date
    document.getElementById('grid-date').valueAsDate = new Date();

    // Login Handling with Credential Verification
    function handleLogin(e) {
      e.preventDefault();
      const usernameInput = document.getElementById('login-username').value.trim();
      const passwordInput = document.getElementById('login-password').value;
      const errorBanner = document.getElementById('login-error');

      // 驗證帳號 teahcer 與密碼 brps2026
      if (usernameInput === 'teahcer' && passwordInput === 'brps2026') {
        errorBanner.classList.add('hidden');
        document.getElementById('module-login').classList.add('hidden');
        document.getElementById('main-app').classList.remove('hidden');
        
        // 清空輸入欄位
        document.getElementById('login-username').value = '';
        document.getElementById('login-password').value = '';
      } else {
        errorBanner.classList.remove('hidden');
      }
    }

    function handleLogout() {
      document.getElementById('main-app').classList.add('hidden');
      document.getElementById('module-login').classList.remove('hidden');
      document.getElementById('login-error').classList.add('hidden');
    }

    // Mock Student Data
    const students = [
      { id: '20240101', name: '陳大文', status: true, note: '' },
      { id: '20240102', name: '李小明', status: true, note: '' },
      { id: '20240103', name: '張偉強', status: true, note: '' },

    // 日期變更處理函數：重置並更新學生狀態
    function handleDateChange() {
      students.forEach(s => {
        s.status = true; // 變更日期後重置為全體「已交」
        s.note = '';     // 清空備註欄位
      });
      renderGrid();
    }

    // Render Student Grid
    function renderGrid() {
      const tbody = document.getElementById('student-grid-body');
      tbody.innerHTML = '';

      students.forEach((s, index) => {
        const tr = document.createElement('tr');
        tr.className = "grid-row hover:bg-blue-50/50 transition border-b border-slate-100";
        tr.setAttribute('tabindex', '0');

        tr.innerHTML = `
          <td class="py-2.5 px-4 font-mono font-bold text-slate-600">${s.id}</td>
          <td class="py-2.5 px-4 font-bold text-slate-800">${s.name}</td>
          <td class="py-2.5 px-4 text-center">
            <button type="button" onclick="toggleStatus(${index})" 
              class="w-24 py-1 rounded-full text-xs font-black transition border ${
                s.status 
                  ? 'bg-emerald-100 border-emerald-300 text-emerald-800 hover:bg-emerald-200' 
                  : 'bg-rose-100 border-rose-300 text-rose-800 hover:bg-rose-200'
              }">
              ${s.status ? '✓ 已交' : '✕ 欠交'}
            </button>
          </td>
          <td class="py-2.5 px-4">
            <input type="text" value="${s.note}" placeholder="填寫欠交原因..." 
              onchange="students[${index}].note = this.value"
              class="w-full bg-transparent hover:bg-white focus:bg-white border border-transparent focus:border-slate-300 rounded px-2 py-1 text-xs focus:outline-none transition">
          </td>
        `;

        // Keyboard navigation (Space to toggle, Arrow Down to move)
        tr.addEventListener('keydown', (e) => {
          if (e.code === 'Space') {
            e.preventDefault();
            toggleStatus(index);
          } else if (e.code === 'ArrowDown') {
            e.preventDefault();
            const next = tr.nextElementSibling;
            if (next) next.focus();
          } else if (e.code === 'ArrowUp') {
            e.preventDefault();
            const prev = tr.previousElementSibling;
            if (prev) prev.focus();
          }
        });

        tbody.appendChild(tr);
      });

      updateCounters();
    }

    function toggleStatus(index) {
      students[index].status = !students[index].status;
      renderGrid();
    }

    function setAllStatus(status) {
      students.forEach(s => s.status = status);
      renderGrid();
    }

    function updateCounters() {
      const submitted = students.filter(s => s.status).length;
      document.getElementById('total-count').innerText = students.length;
      document.getElementById('submitted-count').innerText = submitted;
      document.getElementById('missing-count').innerText = students.length - submitted;
    }

    // Tab Switching Logic
    function switchTab(tab) {
      const gridMod = document.getElementById('module-grid');
      const adminMod = document.getElementById('module-admin');
      const gridBtn = document.getElementById('tab-grid-btn');
      const adminBtn = document.getElementById('tab-admin-btn');

      if (tab === 'grid') {
        gridMod.classList.remove('hidden');
        adminMod.classList.add('hidden');
        gridBtn.className = "px-4 py-1.5 rounded-md text-sm font-medium transition flex items-center gap-2 bg-blue-600 text-white shadow";
        adminBtn.className = "px-4 py-1.5 rounded-md text-sm font-medium text-slate-300 hover:text-white transition flex items-center gap-2";
      } else {
        gridMod.classList.add('hidden');
        adminMod.classList.remove('hidden');
        adminBtn.className = "px-4 py-1.5 rounded-md text-sm font-medium transition flex items-center gap-2 bg-blue-600 text-white shadow";
        gridBtn.className = "px-4 py-1.5 rounded-md text-sm font-medium text-slate-300 hover:text-white transition flex items-center gap-2";
      }
    }

    function switchAdminTab(sub) {
      const pStudents = document.getElementById('admin-panel-students');
      const pClasses = document.getElementById('admin-panel-classes');
      const btnStudents = document.getElementById('admin-subtab-students');
      const btnClasses = document.getElementById('admin-subtab-classes');

      if (sub === 'students') {
        pStudents.classList.remove('hidden');
        pClasses.classList.add('hidden');
        btnStudents.className = "pb-3 text-sm font-bold border-b-2 border-blue-600 text-blue-600 flex items-center gap-2";
        btnClasses.className = "pb-3 text-sm font-medium border-b-2 border-transparent text-slate-500 hover:text-slate-800 flex items-center gap-2";
      } else {
        pStudents.classList.add('hidden');
        pClasses.classList.remove('hidden');
        btnClasses.className = "pb-3 text-sm font-bold border-b-2 border-blue-600 text-blue-600 flex items-center gap-2";
        btnStudents.className = "pb-3 text-sm font-medium border-b-2 border-transparent text-slate-500 hover:text-slate-800 flex items-center gap-2";
      }
    }

    // Init Page
    renderGrid();
