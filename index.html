Google Apps Script 試算表同步設定指南

排查關鍵清單：

  • 網址必須以 /exec 結尾，不能包含 /edit、/dev 或 docs.google.com/spreadsheets。
  • 部署權限「誰可以存取」必須設定為 「所有人 (Anyone)」。
  • 每次修改 GAS 腳本後，必須點擊 「部署」$\rightarrow$「新建部署」 發布新版本。

正確設定步驟：

  1. 開啟您的 Google 試算表，點擊選單列的 「擴充功能」 $\rightarrow$ 「Apps Script」。
  2. 將以下程式碼貼入腳本編輯器中（覆蓋原始 `Code.gs` 的內容）。
  3. 點擊右上方 「部署」 $\rightarrow$ 「新建部署」。
  4. 種類選擇 「網頁應用程式 (Web App)」。
  5. 誰可以存取 (Who has access) 必須選擇：「所有人 (Anyone)」。
  6. 點擊「部署」並授權存取後，複製產生的「網頁應用程式 URL」（需以 /exec 結尾）。
  7. 將複製的網址貼至 Admin 後台的輸入框中，點擊「儲存網址」並按「測試連線」。

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({ 
    status: "online", 
    message: "校園功課管理系統 GAS API 正常服務中！" 
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000); // 避免多人同時寫入衝突
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    
    // 若工作表為空，自動建置表頭
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["班別", "姓名", "欠交日期", "科目", "作業名稱", "登記時間"]);
      sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#e0e7ff");
    }
    
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "無效的 POST 數據" }))
        .setMimeType(ContentService.MimeType.JSON);
    }

    var data = JSON.parse(e.postData.contents);
    var nowStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd HH:mm:ss");
    
    var rowsToAppend = [];
    var list = Array.isArray(data) ? data : [data];
    
    list.forEach(function(row) {
      rowsToAppend.push([
        row.className || row.class || "-", 
        row.studentName || row.name || "-", 
        row.date || "-", 
        row.subject || "-", 
        row.taskName || row.task || "-",
        nowStr
      ]);
    });

    if (rowsToAppend.length > 0) {
      sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, 6).setValues(rowsToAppend);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success", count: rowsToAppend.length }))
      .setMimeType(ContentService.MimeType.JSON);
      
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}
