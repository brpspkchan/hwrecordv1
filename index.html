Google Apps Script 後端程式碼

請貼至 Google 試算表 -> 擴充功能 -> Apps Script 中，並發佈為「Web 應用程式」(所有人皆可存取)。

function doGet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var classes = getSheetData(ss, 'Classes');
  var students = getSheetData(ss, 'Students');
  var logs = getSheetData(ss, 'HomeworkLogs');
  
  return ContentService.createTextOutput(JSON.stringify({
    status: 'success',
    classes: classes,
    students: students,
    logs: logs
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    if (data.action === 'syncAll') {
      var ss = SpreadsheetApp.getActiveSpreadsheet();
      if (data.classes) writeToSheet(ss, 'Classes', data.classes, ['id', 'name']);
      if (data.students) writeToSheet(ss, 'Students', data.students, ['class', 'seat', 'name', 'nameEn', 'status']);
      if (data.logs) writeToSheet(ss, 'HomeworkLogs', data.logs, ['date', 'class', 'subject', 'taskName', 'studentName', 'attendance', 'isMissing', 'remark']);
    }
    return ContentService.createTextOutput(JSON.stringify({ status: 'success' })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: 'error', message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  var headers = values[0];
  var result = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      obj[headers[j]] = row[j];
    }
    result.push(obj);
  }
  return result;
}

function writeToSheet(ss, sheetName, data, headers) {
  var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);
  sheet.clear();
  sheet.appendRow(headers);
  data.forEach(function(item) {
    var row = headers.map(function(h) { return item[h] !== undefined ? item[h] : ''; });
    sheet.appendRow(row);
  });
}
