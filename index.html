學生姓名與 Google 試算表同步診斷器

常見「無紀錄」三大原因排查：

1. 無欠交學生時不觸發 API：只有當學生狀態為「出席」且勾選「欠交作業」時，系統才會發送欠交資料至試算表。

2. 學生姓名包含 BOM 隱形字元：自 Excel 複製的姓名常夾帶隱形符號，可點擊下方「一鍵淨化全校學生姓名」修復。

3. GAS 腳本版本過舊：舊版腳本未包含 `cleanStr()` 淨化處理，請升級 Google Apps Script 代碼。

即時測試發送單一學生至試算表

升級版 Google Apps Script 原始碼 (具姓名淨化與防錯機制)

此升級代碼包含以下修正：

  • 自動淨化姓名 (cleanStr)：去除名字前後隱形字元、UTF-8 BOM 與多餘空格。
  • 相容多種 JSON 欄位命名：`studentName` / `name` / `Name` 均可正確辨識寫入。
  • 支援 doGet 瀏覽器網址測試：直接貼上 `/exec?testName=陳芷晴` 可立刻驗證寫入能力。

完整升級版 Apps Script 腳本：




// 可選填：若非綁定於試算表內，可填入試算表 ID（網址中 d/ 與 /edit 中間的字串）
var SPREADSHEET_ID = "";

function doGet(e) {
if (e && e.parameter && e.parameter.testName) {
var name = cleanStr(e.parameter.testName);
var cls = cleanStr(e.parameter.testClass || "1A");
var res = appendToSheet(

$${ className: cls, studentName: name, date: getTodayStr(), subject: "測試", taskName: "網址測試寫入" }$$

);
return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "已成功寫入測試學生：" + name, count: res }))
.setMimeType(ContentService.MimeType.JSON);
}
return ContentService.createTextOutput(JSON.stringify({
status: "online",
message: "校園功課管理系統 GAS API 正常服務中！"
})).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
var lock = LockService.getScriptLock();
if (!lock.waitLock(10000)) {
return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "系統繁忙，請稍後再試" }))
.setMimeType(ContentService.MimeType.JSON);
}

try {
if (!e || !e.postData || !e.postData.contents) {
return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "無效或空白的 POST 數據" }))
.setMimeType(ContentService.MimeType.JSON);
}

var rawContent = e.postData.contents;
rawContent = rawContent.replace(/^\uFEFF/, '').trim(); // 移除 BOM

var data;
try {
  data = JSON.parse(rawContent);
} catch(err) {
  data = [];
}

var list = Array.isArray(data) ? data : [data];
var count = appendToSheet(list);

return ContentService.createTextOutput(JSON.stringify({ status: "success", count: count }))
  .setMimeType(ContentService.MimeType.JSON);



} catch (error) {
Logger.log("doPost 錯誤: " + error.toString());
return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
.setMimeType(ContentService.MimeType.JSON);
} finally {
lock.releaseLock();
}
}

function appendToSheet(list) {
var ss = (SPREADSHEET_ID && SPREADSHEET_ID.trim() !== "") ? SpreadsheetApp.openById(SPREADSHEET_ID.trim()) : SpreadsheetApp.getActiveSpreadsheet();
if (!ss) throw new Error("無法存取 Google 試算表！請確認開啟權限。");

var sheetName = "欠交紀錄";
var sheet = ss.getSheetByName(sheetName) || ss.insertSheet(sheetName);

if (sheet.getLastRow() === 0) {
sheet.appendRow(

$$"班別", "學生姓名", "欠交日期", "科目", "作業名稱", "系統登記時間"$$

);
sheet.getRange(1, 1, 1, 6).setFontWeight("bold").setBackground("#e0e7ff");
}

var nowStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd HH:mm:ss");
var rowsToAppend = 

;

list.forEach(function(row) {
var sName = cleanStr(row.studentName || row.name || row.Name || "");
var cName = cleanStr(row.className || row.class || row.Class || "-");
var task = cleanStr(row.taskName || row.task || row.Task || "-");
var subj = cleanStr(row.subject || row.Subject || "-");
var dt = cleanStr(row.date || row.Date || getTodayStr());

if (sName !== "") {
  rowsToAppend.push([cName, sName, dt, subj, task, nowStr]);
}



});

if (rowsToAppend.length > 0) {
sheet.getRange(sheet.getLastRow() + 1, 1, rowsToAppend.length, 6).setValues(rowsToAppend);
}
return rowsToAppend.length;
}

function cleanStr(str) {
if (!str) return "";
return String(str)
.replace(/

$$\\uFEFF\\u200B-\\u200D\\uFB00-\\uFDFD$$

/g, "") // 清除零寬度不列印字元與 BOM
.replace(/\s+/g, " ")
.trim();
}

function getTodayStr() {
return Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd");
}
