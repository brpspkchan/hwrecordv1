Google Apps Script 完整雲端資料庫原始碼 (支援多工作表)：

// 綁定指定的 Google 試算表 ID
var SHEET_ID = '1xISwiklE1mBH_d6c0Lh5NxPxXhxy6Q96SKu8qsBd42U';

function getSpreadsheet() {
  try {
    return SpreadsheetApp.openById(SHEET_ID);
  } catch(e) {
    // 降級備用
    return SpreadsheetApp.getActiveSpreadsheet();
  }
}

function doGet(e) {
  var ss = getSpreadsheet();
  var studentsSheet = getOrCreateSheet(ss, "Students", ["班別", "座號", "中文姓名", "英文姓名", "狀態"]);
  var classesSheet = getOrCreateSheet(ss, "Classes", ["班別ID", "班別名稱"]);
  var logsSheet = getOrCreateSheet(ss, "HomeworkLogs", ["日期", "班別", "科目", "作業名稱", "學生姓名", "出席狀態", "是否欠交", "備註", "登記時間"]);

  var students = parseSheetData(studentsSheet, function(row) {
    return { class: String(row[0]), seat: Number(row[1]), name: String(row[2]), nameEn: String(row[3] || ''), status: String(row[4] || 'Active') };
  });

  var classes = parseSheetData(classesSheet, function(row) {
    return { id: String(row[0]), name: String(row[1]) };
  });

  var logs = parseSheetData(logsSheet, function(row) {
    return {
      date: String(row[0]), class: String(row[1]), subject: String(row[2]), taskName: String(row[3]),
      studentName: String(row[4]), attendance: String(row[5]), isMissing: row[6] === 'true' || row[6] === true,
      remark: String(row[7] || '')
    };
  });

  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    students: students,
    classes: classes,
    logs: logs
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = getSpreadsheet();
    if (!e || !e.postData || !e.postData.contents) {
      return ContentService.createTextOutput(JSON.stringify({ status: "error", message: "無效數據" })).setMimeType(ContentService.MimeType.JSON);
    }

    var data = JSON.parse(e.postData.contents);
    var nowStr = Utilities.formatDate(new Date(), "GMT+8", "yyyy-MM-dd HH:mm:ss");

    if (data.students) {
      updateSheetData(ss, "Students", ["班別", "座號", "中文姓名", "英文姓名", "狀態"], 
        data.students.map(function(s) { return [s.class, s.seat, s.name, s.nameEn || '', s.status || 'Active']; }));
    }
    if (data.classes) {
      updateSheetData(ss, "Classes", ["班別ID", "班別名稱"], 
        data.classes.map(function(c) { return [c.id, c.name]; }));
    }
    if (data.logs) {
      updateSheetData(ss, "HomeworkLogs", ["日期", "班別", "科目", "作業名稱", "學生姓名", "出席狀態", "是否欠交", "備註", "登記時間"], 
        data.logs.map(function(l) { return [l.date, l.class, l.subject, l.taskName, l.studentName, l.attendance, l.isMissing ? 'true' : 'false', l.remark || '', nowStr]; }));
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", message: "雲端資料庫已同步" })).setMimeType(ContentService.MimeType.JSON);
  } catch (error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON);
  } finally {
    lock.releaseLock();
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e0e7ff");
  }
  return sheet;
}

function parseSheetData(sheet, mapFn) {
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  var result = [];
  for (var i = 1; i < values.length; i++) {
    if (values[i][0] !== "") {
      result.push(mapFn(values[i]));
    }
  }
  return result;
}

function updateSheetData(ss, sheetName, headers, rows) {
  var sheet = getOrCreateSheet(ss, sheetName, headers);
  sheet.clearContents();
  sheet.appendRow(headers);
  sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e0e7ff");
  if (rows.length > 0) {
    sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
  }
}
